VALIDAÇÃO RÁPIDA

Query 1: Ver candidato com múltiplas zonas
SELECT p.nome, m.nome, vm.nr_turno, vm.total_votos,
  (SELECT COUNT(DISTINCT zona_id) FROM votos_zona vz
   WHERE vz.candidatura_id = vm.candidatura_id
     AND vz.municipio_id = vm.municipio_id) as total_zonas
FROM votos_municipio vm
JOIN candidaturas c ON c.id = vm.candidatura_id
JOIN pessoas p ON p.id = c.pessoa_id
JOIN municipios m ON m.id = vm.municipio_id
LIMIT 1;

Resultado esperado: uma linha com total_zonas > 1

Query 2: Validar somas conferem
SELECT vm.candidatura_id, vm.total_votos as municipio,
  (SELECT SUM(total_votos) FROM votos_zona vz
   WHERE vz.candidatura_id = vm.candidatura_id
     AND vz.municipio_id = vm.municipio_id) as zona_soma,
  CASE WHEN vm.total_votos = (SELECT SUM(total_votos) FROM votos_zona vz
                               WHERE vz.candidatura_id = vm.candidatura_id
                                 AND vz.municipio_id = vm.municipio_id)
       THEN '✅' ELSE '❌' END
FROM votos_municipio vm LIMIT 5;

Resultado esperado: todas as linhas com ✅
